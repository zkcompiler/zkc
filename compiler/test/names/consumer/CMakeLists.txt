cmake_minimum_required(VERSION 3.21)
project(ZkcNamesConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_DISABLE_FIND_PACKAGE_MLIR TRUE)
if(ZKC_PRELOAD_WRONG_UTF8PROC)
  add_library(utf8proc::utf8proc INTERFACE IMPORTED)
  set(utf8proc_VERSION "2.11.0")
endif()
if(ZKC_EXPECT_MISSING)
  find_package(ZkcCompiler QUIET COMPONENTS Syntax)
  if(ZkcCompiler_FOUND OR ZkcCompiler_Source_FOUND OR TARGET Zkc::Source OR TARGET Zkc::Syntax)
    message(FATAL_ERROR "Source accepted a missing or incompatible NFC dependency")
  endif()
  # A failed optional Source request must leave LLVM-only components usable.
  find_package(ZkcCompiler REQUIRED COMPONENTS Support)
  return()
endif()
find_package(ZkcCompiler REQUIRED COMPONENTS Syntax)
find_package(ZkcCompiler REQUIRED COMPONENTS Source Syntax Support)
if(TARGET MLIRIR OR MLIR_FOUND)
  message(FATAL_ERROR "Source must discover without MLIR")
endif()
add_executable(names-consumer main.cpp)
target_link_libraries(names-consumer PRIVATE Zkc::Syntax)
enable_testing()
add_test(NAME installed-names COMMAND names-consumer)
