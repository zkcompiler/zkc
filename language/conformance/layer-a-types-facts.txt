// The values of a multilinear polynomial in v variables on the Boolean cube, first variable
// most significant.
pub type Table<v: nat> = [F; 2^v]

// The sums of the two halves.
pub def sums<v: nat>(table: Table<v + 1>) -> (F, F):

// One round of a sumcheck over a polynomial of degree at most Degree in each variable.
pub interface Round:
    nat Degree
    type State<v: nat>
    type Message
    def message<v: nat>(state: State<v + 1>) -> Message
    def bind<v: nat>(state: State<v + 1>, r: F) -> State<v>
    def boundary(m: Message) -> F
    def evaluate(m: Message, r: F) -> F

    spec def polynomial(m: Message) -> Poly<F, Degree>

    law ends(m: Message): boundary(m) == value(polynomial(m), 0) + value(polynomial(m), 1)

    nat Degree = 1

    type Message = (F, F)
    def message(state): sums(state)
    def bind(state, r): fix_first(state, r)
    def boundary(m): m.0 + m.1
    def evaluate(m, r): (1 - r) * m.0 + r * m.1
    spec def table(state): extension(state)
    spec def polynomial(m): line(m.0, m.1)

// The sum of p over the Boolean cube, after its first k variables are fixed to point, is value.
pub relation PartialSum<D: nat, n: nat, k: nat>(p: MvPoly<F, n + k, D>, point: [F; k], value: F):
    value == ∑ b ∈ cube<F, n>(), evaluate_at(p, point ++ b)

// k + rounds variables fixed.
pub protocol Reduce<R: Round, v: nat, k: nat, rounds: nat> roles(P, V)(
    spec p: MvPoly<F, v + rounds + k, R::Degree>,
    state: R::State<v + rounds> @P,
    prefix: [F; k] @(P, V),
    claim: F @(P, V)

pub relation DLog<G: PrimeOrderGroup>(base: G, point: G; scalar: G::Scalar):
    point == base * scalar

pub protocol Schnorr<G: PrimeOrderGroup> roles(P, V)(
    base: G @(P, V),
    point: G @(P, V),
    scalar: G::Scalar @P
):
    nonce @P <- sample G::Scalar
    commitment <- send base * nonce to V
    challenge <- draw G::Scalar from V to P
    response <- send nonce + challenge * scalar to V
    guard @V base * response == commitment + point * challenge

// The tree function of special soundness: two accepting runs with one commitment and different
// challenges. The tree guarantees the condition of the guard.
pub def response_difference<G: PrimeOrderGroup>(
    base: G, point: G, runs: Schnorr<G>::Runs<2>
) -> G::Scalar:
    gap = runs[0].challenge - runs[1].challenge
    return (runs[0].response - runs[1].response) / gap

// The simulator of the honest verifier's view: its inputs, its coins, and one uniform scalar.
pub def simulate<G: PrimeOrderGroup>(
    inputs: Schnorr<G>::Inputs<V>,
    coins: Schnorr<G>::Coins<V>,
    response: G::Scalar
) -> Schnorr<G>::Received<V>:
    Schnorr<G>::Received<V>{
        commitment = inputs.base * response - inputs.point * coins.challenge,
        response
    }

// The simulator of a derived scheme: it chooses the challenge, and the experiment programs the
// random oracle to return it. Its randomness is a pair: the challenge and the response.
pub def simulate_programmed<G: PrimeOrderGroup>(
    inputs: Schnorr<G>::Inputs<V>,
    randomness: (G::Scalar, G::Scalar)
) -> (Schnorr<G>::Received<V>, Schnorr<G>::Coins<V>):
    challenge = randomness.0
    response = randomness.1
    return (
        Schnorr<G>::Received<V>{
            commitment = inputs.base * response - inputs.point * challenge,
            response
        },
        Schnorr<G>::Coins<V>{challenge}
    )

// `rounds` rounds of sumcheck: a claim about p with k variables fixed becomes a claim with
// k + rounds variables fixed.
pub protocol Reduce<R: Round, v: nat, k: nat, rounds: nat> roles(P, V)(
    spec p: MvPoly<F, v + rounds + k, R::Degree>,
    state: R::State<v + rounds> @P,
    prefix: [F; k] @(P, V),
    claim: F @(P, V)
) -> (point: [F; k + rounds] @(P, V), value: F @(P, V), residual: R::State<v> @P):
    mut s = state
    mut running = claim
    for i in 0..rounds collect (challenges = challenge):
        round <- send R::message(s) to V
        guard @V running == R::boundary(round)
        challenge <- draw F from V to P
        s := R::bind(s, challenge)
        running := R::evaluate(round, challenge)
    return (prefix ++ challenges, running, s)

    ensures R::table(residual) == restrict_prefix(p, point)

    through PartialSum(p, prefix ++ challenges, running) at i
    invariant R::table(s) == restrict_prefix(p, prefix ++ challenges) at i
    charge sites.challenge <= R::Degree / F::Order

    claim == ∑ v ∈ values, v

    values: Table<n> @(P, V),
    claim: F @(P, V)
):
    reduced <- Reduce<Pairs, rounds = n>(values, [], claim, spec p = extension(values))
    guard @V reduced.value == evaluate(values, reduced.point)

use zkc::curves::bls12_381::{Fr, G1}
use zkc::option::{Option}

pub struct Pair<T>:
    pub left: T
    pub right: T

pub linear struct Ticket<F: Field>:
    point: [F]
    value: F

pub enum Side:
    Left
    Right

pub def pick<T>(pair: Pair<T>, side: Side) -> T:
    Pair{left, right} = pair
    mut chosen = left
    match side:
        case Left:
            ()
        case Right:
            chosen := right
    return chosen

pub def swap<T>(pair: Pair<T>) -> Pair<T>: {pair with left = pair.right, right = pair.left}

// `+` on `Pair<F>` is this function: the fixed name `add` in the home module of `Pair`. With
// `zero`, the type also has `∑`.
pub def add<F: Field>(a: Pair<F>, b: Pair<F>) -> Pair<F>:
    Pair{left = a.left + b.left, right = a.right + b.right}

pub def zero<F: Field>() -> Pair<F>: Pair{left = 0, right = 0}

pub def twice<F: Field>(a: Pair<F>) -> Pair<F>: a + a

pub def redeem<F: Field>(ticket: Ticket<F>) -> F:
    Ticket{point = _, value} = ticket
    return value

pub def total<F: Field>(values: [F]) -> F: ∑ x ∈ values, x

// A read of a collection whose length is a run-time value: the guard gives the fact.
pub def first<F: Field>(xs: [F]) -> F:
    guard 0 < length(xs)
    return xs[0]

// The same read without a stop: the condition of an `if` gives the fact to its first arm.
pub def first_or_none<F: Field>(xs: [F]) -> Option<F>:
    if 0 < length(xs) then Some(xs[0]) else None

// The index of a loop over `0..length(xs)` is below the length in its body.
pub def weighted<F: Field>(xs: [F], weight: F) -> F:
    mut acc: F = 0
    for i in 0..length(xs):
        acc := acc * weight + xs[i]
    return acc

    half = (2 * n) / 2
    return (xs[0..half], xs[half..2 * half])

    (∑ p ∈ points, p) + G::generator() * G::Scalar::from_nat(n) + 0

pub relation Inner<F: Field, n: nat>(a: [F; n], b: [F; n], c: F): c == ⟪a, b⟫

pub relation Ascending<n: nat>(xs: [byte; n + 1]): ∀ i ∈ 0..n, xs[i] <= xs[i + 1]

pub spec def checked(a: [Fr; 4], b: [Fr; 4], c: Fr, xs: [byte; 5]) -> bool:
    Inner(a, b, c) and Ascending(xs)

pub spec def birthday(queries: nat, bits: nat) -> rational: queries^2 / 2^(bits + 1)

// The functions above at one field: a definition without static parameters is its own instance.
pub def demo(pair: Pair<Fr>, side: Side, ticket: Ticket<Fr>, values: [Fr], points: [G1; 4]) -> G1:
    chosen = pick(swap(twice(pair)), side)

pub protocol Ask<F: Field> roles(O, R)() -> (value: F @(O, R), nonce: nonzero<F> @O):
    nonce @O <- sample nonzero<F>
    value <- send nonce * nonce to R
    return (value, nonce)

// A callee with roles (O, R) applied by name; a pattern binder named by an attribute; a loop
// with a run-time count; a private input; a send to two receivers.
pub protocol Relay<F: Field, Max: nat> roles(D, P, V)(
    count: index<Max + 1> @(P, V),
    seed: [byte; 32] @V private
) -> total: F @(P, V):
    mut running: F @(P, V) = 0
    for _ in 0..count:
        #[site(step)]
        (value, _) <- Ask<F>() @(O = P, R = V)
        running := running + value
    share @D <- sample F
    told <- send share to P, V
    return running + told

pub enum Witness<G: PrimeOrderGroup>:
    Left(G::Scalar)
    Right(G::Scalar)

// A proof that the prover knows the discrete logarithm of x0 or of x1. Both matches are
// private to P: no message tells V which alternative P holds.
pub protocol Or<G: PrimeOrderGroup> roles(P, V)(
    base: G @(P, V), x0: G @(P, V), x1: G @(P, V),
    witness: Witness<G> @P
):
    nonce @P <- sample G::Scalar
    simulated_challenge @P <- sample G::Scalar
    simulated_response @P <- sample G::Scalar
    mut a0 = base * nonce
    mut a1 = base * nonce
    match witness:
        case Witness::Left(_):
            a1 := base * simulated_response - x1 * simulated_challenge
        case Witness::Right(_):
            a0 := base * simulated_response - x0 * simulated_challenge
    commitments <- send (a0, a1) to V
    challenge <- draw G::Scalar from V to P
    mut e0 = simulated_challenge
    mut e1 = simulated_challenge
    mut z0 = simulated_response
    mut z1 = simulated_response
    match witness:
        case Witness::Left(w):
            e0 := challenge - simulated_challenge
            z0 := nonce + e0 * w
            z1 := w
        case Witness::Left(v):
            ()
        case Witness::Right(w):
            e1 := challenge - simulated_challenge
            z1 := nonce + e1 * w
