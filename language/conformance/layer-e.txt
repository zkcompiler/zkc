Conformance tests of layer E (constraint systems). NOT FROZEN: the layer is carried,
and its rules can still change; an outcome that rests on a pending change says so in its note.
One block per test: `=== ID | expect | note`, then the program. See README.md.

=== N-E1 | accept | a constraint system as one function over an algebra; the stated degree is a closed law, decided true by evaluation at the degree interpretation
module t
type F = primitive("koala-bear")
struct Bound:
    pub degree: nat
interface Algebra:
    type T
    def constant(c: F) -> T
    def add(a: T, b: T) -> T
    def sub(a: T, b: T) -> T
    def mul(a: T, b: T) -> T
module Over: Algebra:
    type T = F
    def constant(c): c
    def add(a, b): a + b
    def sub(a, b): a - b
    def mul(a, b): a * b
def larger(a: nat, b: nat) -> nat: if a < b then b else a
module Degrees: Algebra:
    type T = Bound
    def constant(c): Bound{degree = 0}
    def add(a, b): Bound{degree = larger(a.degree, b.degree)}
    def sub(a, b): Bound{degree = larger(a.degree, b.degree)}
    def mul(a, b): Bound{degree = a.degree + b.degree}
interface System:
    nat Width
    nat Degree
    def constraint<X: Algebra>(row: [X::T; Width]) -> X::T
    law degree(): constraint<Degrees>([Bound{degree = 1}; Width]).degree <= Degree
module Square: System:
    nat Width = 2
    nat Degree = 2
    def constraint(row): row[0] * row[0] - row[1]
