    base: G @(P, V),

// The tree function of special soundness: two accepting runs with one commitment and different
// challenges. The tree guarantees the condition of the guard.
pub def response_difference<G: PrimeOrderGroup>(
    base: G, point: G, runs: Schnorr<G>::Runs<2>
) -> G::Scalar:
    gap = runs[0].challenge - runs[1].challenge
    guard gap != 0
    return (runs[0].response - runs[1].response) / gap

pub def simulate<G: PrimeOrderGroup>(
    inputs: Schnorr<G>::Inputs<V>,
    coins: Schnorr<G>::Coins<V>,
    response: G::Scalar
) -> Schnorr<G>::Received<V>:
    Schnorr<G>::Received<V>{
        commitment = inputs.base * response - inputs.point * coins.challenge,
        response
    }

// The simulator of a derived scheme: it chooses the challenge, and the experiment programs the
// random oracle to return it. Its randomness is a pair: the challenge and the response.
pub def simulate_programmed<G: PrimeOrderGroup>(
    inputs: Schnorr<G>::Inputs<V>,
    randomness: (G::Scalar, G::Scalar)
) -> (Schnorr<G>::Received<V>, Schnorr<G>::Coins<V>):
    challenge = randomness.0
    response = randomness.1
    return (
        Schnorr<G>::Received<V>{
            commitment = inputs.base * response - inputs.point * challenge,
            response
        },
        Schnorr<G>::Coins<V>{challenge}
    )

// The values of a multilinear polynomial in v variables on the Boolean cube, first variable
// most significant.
pub type Table<v: nat> = [F; 2^v]

// The sums of the two halves.
pub def sums<v: nat>(table: Table<v + 1>) -> (F, F):
    (sum(table[0..2^v]), sum(table[2^v..2^(v + 1)]))

// One round of a sumcheck over a polynomial of degree at most Degree in each variable.
pub interface Round:
    nat Degree
    type State<v: nat>
    type Message
    def message<v: nat>(state: State<v + 1>) -> Message
    def bind<v: nat>(state: State<v + 1>, r: F) -> State<v>
    def boundary(m: Message) -> F
    def evaluate(m: Message, r: F) -> F
    spec def table<v: nat>(state: State<v>) -> MvPoly<F, v, Degree>
    spec def polynomial(m: Message) -> Poly<F, Degree>
    law sent<v: nat>(s: State<v + 1>): polynomial(message(s)) == first_variable(table(s))
    law bound<v: nat>(s: State<v + 1>, r: F): table(bind(s, r)) == restrict(table(s), r)
    law ends(m: Message): boundary(m) == value(polynomial(m), 0) + value(polynomial(m), 1)
    law at_point(m: Message, r: F): evaluate(m, r) == value(polynomial(m), r)

// Rounds over a multilinear table: the sums of its two halves.
pub module Pairs: Round:
    nat Degree = 1
    type State<v: nat> = Table<v>
    type Message = (F, F)
    def message(state): sums(state)
    def bind(state, r): fix_first(state, r)
    def boundary(m): m.0 + m.1
    def evaluate(m, r): (1 - r) * m.0 + r * m.1
    spec def table(state): extension(state)
    spec def polynomial(m): line(m.0, m.1)

// The sum of p over the Boolean cube, after its first k variables are fixed to point, is value.
pub relation PartialSum<D: nat, n: nat, k: nat>(p: MvPoly<F, n + k, D>, point: [F; k], value: F):
    value == ∑ b ∈ cube<F, n>(), evaluate_at(p, point ++ b)

// `rounds` rounds of sumcheck: a claim about p with k variables fixed becomes a claim with
// k + rounds variables fixed.
pub protocol Reduce<R: Round, v: nat, k: nat, rounds: nat> roles(P, V)(
    spec p: MvPoly<F, v + rounds + k, R::Degree>,
    state: R::State<v + rounds> @P,
    prefix: [F; k] @(P, V),
    claim: F @(P, V)
) -> (point: [F; k + rounds] @(P, V), value: F @(P, V), residual: R::State<v> @P):
    mut s = state
    mut running = claim
    for i in 0..rounds collect (challenges = challenge):
        round <- send R::message(s) to V
        guard @V running == R::boundary(round)
        challenge <- draw F from V to P
        s := R::bind(s, challenge)
        running := R::evaluate(round, challenge)
    return (prefix ++ challenges, running, s)

spec for Reduce:
    requires R::table(state) == restrict_prefix(p, prefix)
    ensures R::table(residual) == restrict_prefix(p, point)
    reduces @V PartialSum(p, prefix, claim) to PartialSum(p, point, value)
    through PartialSum(p, prefix ++ challenges, running) at i
    invariant R::table(s) == restrict_prefix(p, prefix ++ challenges) at i
    charge sites.challenge <= R::Degree / F::Order

    claim == ∑ v ∈ values, v

pub protocol Sumcheck<n: nat> roles(P, V)(
    values: Table<n> @(P, V),
    claim: F @(P, V)

    reduced <- Reduce<Pairs, rounds = n>(values, [], claim, spec p = extension(values))
    guard @V reduced.value == evaluate(values, reduced.point)

section <G: PrimeOrderGroup>:
    // P = g^a h^b u^<a, b>, written additively.
    def commitment<m: nat>(
        g: [G; 2^m],
        h: [G; 2^m],
        u: G,
        a: [G::Scalar; 2^m],
        b: [G::Scalar; 2^m]
    ) -> G:
        ⟪g, a⟫ + ⟪h, b⟫ + u * ⟪a, b⟫

    relation InnerProductRelation<n: nat>(
        g: [G; 2^n], h: [G; 2^n], u: G, target: G; a: [G::Scalar; 2^n], b: [G::Scalar; 2^n]
    ):
        target == commitment(g, h, u, a, b)

    // A non-trivial discrete-logarithm relation among the generators.
    relation DiscreteLogRelation<n: nat>(
        g: [G; 2^n], h: [G; 2^n], u: G; r: [G::Scalar; 2^n], s: [G::Scalar; 2^n], t: G::Scalar
    ):
        ⟪g, r⟫ + ⟪h, s⟫ + u * t == 0 and (∃ c ∈ r ++ s ++ [t], c != 0)

    protocol InnerProduct<n: nat> roles(P, V)(
        g: [G; 2^n] @(P, V),
        h: [G; 2^n] @(P, V),
        u: G @(P, V),
        target: G @(P, V),
        a: [G::Scalar; 2^n] @P,
        b: [G::Scalar; 2^n] @P
    ):
        mut gk = g
        mut hk = h
        mut pk = target
        mut ak = a
        mut bk = b
        for k in 0..n:
            left <- send commitment(high(gk), low(hk), u, low(ak), high(bk)) to V
            right <- send commitment(low(gk), high(hk), u, high(ak), low(bk)) to V
            x <- draw nonzero<G::Scalar> from V to P
            y = 1 / x
            gk := fold(gk, y, x)
            hk := fold(hk, x, y)
            pk := left * x^2 + pk + right * y^2
            ak := fold(ak, x, y)
            bk := fold(bk, y, x)
        a_last <- send ak[0] to V
        b_last <- send bk[0] to V
        guard @V pk == gk[0] * a_last + hk[0] * b_last + u * (a_last * b_last)

    // formal library's, and a goal that uses this clause is reported as resting on it.

        through InnerProductRelation(gk, hk, u, pk; ak, bk) at k

    def scalars<n: nat>(x: [G::Scalar; n], y: [G::Scalar; n]) -> [G::Scalar; 2^n]:
        [(∏ j ∈ 0..n, if (i / 2^(n - j - 1)) % 2 == 1 then x[j] else y[j]) for i in 0..2^n]

    // and the challenges and computes the folded generators once, at the end.
    protocol InnerProductOnce<n: nat> roles(P, V)(

        h: [G; 2^n] @(P, V),
        u: G @(P, V),
        target: G @(P, V),
        a: [G::Scalar; 2^n] @P,
        b: [G::Scalar; 2^n] @P

        mut hk @P = h
        mut ak = a
        mut bk = b
