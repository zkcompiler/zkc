cmake_minimum_required(VERSION 3.21)
project(ZkcEvmConsumer LANGUAGES C CXX)
find_package(ZkcCompiler REQUIRED CONFIG COMPONENTS EvmTarget)
if(TARGET Zkc::CpuTarget OR TARGET Zkc::Driver)
  message(FATAL_ERROR "EVM target imported CPU target or driver")
endif()
add_executable(evm-consumer main.cpp)
target_link_libraries(evm-consumer PRIVATE Zkc::EvmTarget)
target_compile_options(evm-consumer PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME evm-consumer COMMAND evm-consumer)
