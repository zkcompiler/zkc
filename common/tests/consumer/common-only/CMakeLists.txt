cmake_minimum_required(VERSION 3.21)
project(ZkcCommonConsumer LANGUAGES C CXX)
find_package(ZkcCompiler REQUIRED CONFIG COMPONENTS Compiler)
if(TARGET Zkc::CpuTarget OR TARGET Zkc::EvmTarget OR TARGET Zkc::Driver)
  message(FATAL_ERROR "Common compiler imported a machine target or driver")
endif()
add_executable(common-consumer ../compiler.cpp)
target_link_libraries(common-consumer PRIVATE Zkc::Compiler)
target_compile_options(common-consumer PRIVATE -Wall -Wextra -Werror)
get_target_property(include_dirs Zkc::Support INTERFACE_INCLUDE_DIRECTORIES)
foreach(directory IN LISTS include_dirs)
  if(EXISTS "${directory}/zkc/Compiler")
    file(GLOB headers "${directory}/zkc/Compiler/*.h" "${directory}/zkc/Transforms/*.h")
    foreach(header IN LISTS headers)
      file(RELATIVE_PATH relative "${directory}" "${header}")
      string(SHA256 identifier "${relative}")
      set(unit "${CMAKE_CURRENT_BINARY_DIR}/headers/${identifier}.cpp")
      file(GENERATE OUTPUT "${unit}" CONTENT "#include <${relative}>\n")
      list(APPEND units "${unit}")
    endforeach()
  endif()
endforeach()
if(NOT units)
  message(FATAL_ERROR "Installed common public headers were not found")
endif()
add_library(common-public-headers OBJECT ${units})
target_link_libraries(common-public-headers PRIVATE Zkc::Compiler)
target_compile_options(common-public-headers PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME common-consumer COMMAND common-consumer)
